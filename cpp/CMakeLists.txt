cmake_minimum_required(VERSION 3.16)
project(EmbeddedJournal LANGUAGES CXX)
enable_testing()
add_executable(core_tests tests/test_core.cpp)
target_include_directories(core_tests PRIVATE include)
target_compile_features(core_tests PRIVATE cxx_std_11)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(core_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME core_tests COMMAND core_tests)
